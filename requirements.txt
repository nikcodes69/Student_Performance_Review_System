# Student Performance Review & Prediction System
# Tested against Python 3.13 (this machine did not have 3.11 installed;
# these are the exact versions that installed cleanly via prebuilt wheels
# on Python 3.13 -- see README for details).
# Install with:  pip install -r requirements.txt

# --- Web application framework ---
streamlit==1.64.0

# --- Google Sign-In (st.login/st.logout/st.user) -- see
# modules/auth.py's authenticate_with_google(). Only used when an
# [auth]/[auth.google] section is configured in Streamlit secrets;
# the rest of the app works unchanged without it.
Authlib==1.8.0

# --- Data handling ---
# pandas is pinned to 2.2.3, not the newer 3.0.5, because 3.0.5's
# compiled wheel was flagged by Windows Smart App Control on the
# development machine (an OS-level, reputation-based block on the
# specific compiled binary -- unrelated to this project's own code, and
# not something a "trusted publisher" pip install can route around).
# 2.2.3's differently-built wheel was not flagged. Every test in this
# project (199 as of this pin) passes identically under 2.2.3.
pandas==2.2.3
numpy==2.5.3

# --- Excel (.xlsx) read/write -- exporting any table, and reading a
# bulk-import spreadsheet, both go through pandas, which needs this
# engine installed for .xlsx specifically (CSV needs no extra package).
openpyxl==3.1.5

# --- Charts (analytics dashboard, ML result plots) ---
plotly==7.1.0

# --- Machine learning ---
scikit-learn==1.9.1
joblib==1.6.0

# --- Password hashing ---
bcrypt==5.0.0

# --- Cloud database (Turso/libSQL) -- used when TURSO_DATABASE_URL and
# TURSO_AUTH_TOKEN are configured in Streamlit secrets; falls back to a
# local SQLite file otherwise. See database/db_setup.py's get_connection().
libsql==0.1.11

# --- PDF report card generation ---
reportlab==5.0.1

# --- Testing ---
pytest==9.1.1

# --- Notebook support (for ml/model_training.ipynb) ---
notebook==7.6.2
ipykernel==7.3.0
